### 1. Essential Python Libraries for Windows
To build this stack, you will need the following packages installable via `pip`:
* **`psutil`**: Cross-platform process and system utilities module (used to enumerate running Windows processes, check command-line arguments, open files, and parent-child trees).
* **`yara-python`**: Python bindings for YARA[1], allowing you to compile signature files and scan target files or process memory spaces.
* **`pefile`**: (Optional) For static analysis of Windows PE headers (imports, sections, entropy) if you decide to inspect executables on disk before they launch[2].

```bash
pip install psutil yara-python pefile
```

---

### 2. Conceptual Architecture
1. **Process Monitor (Collector):** Periodically polls active Windows processes using `psutil` or hooks into Windows Event Tracing (ETW) / WMI. It extracts metadata: executable path, command line arguments, spawned child processes, and loaded DLLs.
2. **Heuristic Filter:** Evaluates metadata attributes for anomaly indicators (e.g., a script interpreter like `powershell.exe` or `cmd.exe` executing encoded payloads, execution running out of temporary directories like `%TEMP%` or `AppData\Roaming`, or unusual process trees like `winword.exe` spawning `cmd.exe`).
3. **YARA Scanner:** Maps flagged process binaries or memory regions against a compiled set of YARA `.yar` rules to look for known shellcode patterns, packer signatures, or malicious strings[1].

### 3. Example Code Template
The following Python script demonstrates a lightweight loop that monitors active Windows processes, applies **heuristic checks** (suspicious execution paths), and uses **YARA** to scan the process binary on disk if available.

```python
import os
import time
import psutil
import yara

# Define simple heuristic suspicious paths or keywords on Windows
SUSPICIOUS_PATHS = ["\\appdata\\local\\temp\\", "\\public\\", "\\downloads\\"]
SUSPICIOUS_NAMES = ["mimikatz", "nc.exe", "netcat", "psexec"]

# Compile a basic YARA ruleset (or load from a file)
# In production, use yara.compile(filepaths={'my_rules': 'path/to/rules.yar'})
yara_rules = yara.compile(
    source="""
    rule Suspicious_API_Strings {
        strings:
            $s1 = "VirtualAllocEx" nocase
            $s2 = "WriteProcessMemory" nocase
            $s3 = "CreateRemoteThread" nocase
        condition:
            all of them
    }
"""
)


def evaluate_heuristics(proc):
  """Applies heuristic rules based on process metadata."""
  score = 0
  flags = []

  try:
    name = proc.info["name"] or ""
    exe_path = (proc.info["exe"] or "").lower()
    cmdline = " ".join(proc.info["cmdline"] or []).lower()

    # Heuristic 1: Known malicious utility names masquerading or used maliciously
    if any(bad in name.lower() for bad in SUSPICIOUS_NAMES):
      score += 50
      flags.append(f"Suspicious process name: {name}")

    # Heuristic 2: Execution running out of typical drop zones (e.g., Temp)
    if any(path in exe_path for path in SUSPICIOUS_PATHS):
      score += 30
      flags.append(f"Running from suspicious directory: {exe_path}")

    # Heuristic 3: Suspicious command-line flags (e.g., hidden powershell)
    if "powershell" in name.lower() and (
        "-enc" in cmdline or "-windowstyle hidden" in cmdline
    ):
      score += 40
      flags.append("Obfuscated or hidden PowerShell execution")

  except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
    pass

  return score, flags


def scan_binary_with_yara(file_path):
  """Scans a file path on disk using compiled YARA rules."""
  if not file_path or not os.path.exists(file_path):
    return []
  try:
    matches = yara_rules.match(file_path)
    return [match.rule for match in matches]
  except Exception:
    # Access denied or file locked by OS
    return []


def monitor_processes():
  print("[*] Starting Windows Process Monitor with Heuristics & YARA...")
  seen_pids = set()

  while True:
    current_pids = set()
    for proc in psutil.process_iter(attrs=["pid", "name", "exe", "cmdline"]):
      try:
        pid = proc.info["pid"]
        current_pids.add(pid)

        # Process newly spawned pids to avoid log spamming
        if pid not in seen_pids:
          seen_pids.add(pid)

          # 1. Run Heuristics
          h_score, h_flags = evaluate_heuristics(proc)

          # 2. Run YARA Scan on the executable path
          exe_path = proc.info.get("exe")
          yara_matches = scan_binary_with_yara(exe_path)

          # 3. Decision Matrix
          if h_score >= 40 or yara_matches:
            print(f"\n[!] ALERT: Suspicious Process Detected (PID: {pid})")
            print(f"    Name: {proc.info.get('name')}")
            print(f"    Path: {exe_path}")
            print(f"    Heuristic Score: {h_score} | Flags: {h_flags}")
            if yara_matches:
              print(f"    YARA Matches: {yara_matches}")

      except (psutil.NoSuchProcess, psutil.AccessDenied):
        continue

    # Cleanup dead PIDs from tracking set
    seen_pids = seen_pids.intersection(current_pids)
    time.sleep(2)


if __name__ == "__main__":
  monitor_processes()
```
